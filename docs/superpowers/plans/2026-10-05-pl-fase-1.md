# P&L mensual (fase 1) — Plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** P&L mensual por sucursal y consolidado con los renglones del modelo de Dodo, real contra meta, con detalle por renglón y export a PDF/Excel.

**Architecture:** Migración 0025 agrega la cuenta del P&L a proveedores, facturas y caja chica, y las tablas de captura (parámetros con vigencia, plantilla y ajustes de nómina, gastos del mes, metas). El cálculo vive en `lib/finanzas/` como funciones puras con pruebas (parámetros vigentes, nómina, comisiones, prorrateo, armado del P&L, hojas de Excel). `app/finanzas/datos.ts` carga los datos del mes, arma la entrada de cada sucursal y llama al cálculo; las páginas solo muestran.

**Tech Stack:** Next.js 15 (App Router, server actions), React 19, TypeScript estricto, Supabase (Postgres + RLS), `node --experimental-strip-types --test`.

**Spec:** `docs/superpowers/specs/2026-10-05-pl-fase-1-design.md`

## Global Constraints

- Montos del P&L **sin IVA**, criterio **devengado**; cálculos en centavos (`Math.round(n * 100) / 100`), % con un decimal en pantalla.
- Regalías **5 %** de la venta neta. Rappi: **0 % desde 2026-10-01**, **18 % desde 2027-01-01**, sobre la venta del canal agregador **con IVA**.
- Comisiones de cobro sobre la venta **con IVA** del medio, sin la parte del agregador: dLocal 3.5 % (Oxxo 5 %), NetPay 1.8 % (Amex/internacional 3.6 %); participación Oxxo y Amex como parámetros (0 de inicio).
- Monto de factura en el P&L = `subtotal − descuento`; notas de crédito (`tipo = 'egreso'`) restan; canceladas no cuentan; sin sucursal = oficina central.
- Permisos: **admin** escribe; **admin y corporativo** leen; gerente/encargado/operación no ven Finanzas.
- Los módulos de `lib/` se importan entre sí **solo como tipos** (`import type`, sin extensión); los tests importan valores con `.ts`.
- Archivos CRLF: editar con Edit; reemplazos en scripts de node con `s.replace(a, () => b)` (nunca un string con `$` como reemplazo).
- Cada prueba nueva de `lib/` se agrega al script `test` de `package.json`.
- Textos de la UI en español de México, tarjetas/listas sin scroll horizontal (clases `lista-filas`, `tabla-tarjetas` existentes).

## Review Focus

1. **Doble conteo de comisiones:** la venta de Rappi aparece también en `tarjeta`/`en_linea` del día; la base de dLocal/NetPay debe restar la parte del agregador (prueba en Tarea 4).
2. **Vigencias a mitad de mes:** un parámetro o puesto que cambia el día 15 debe aplicar por día (comisiones, regalías) o con la plantilla vigente al cierre del mes (nómina) — prueba en Tareas 2–4.
3. **Prorrateo:** sin ventas en ninguna sucursal, la oficina no se reparte (queda solo en el consolidado); con una sola sucursal con venta, se le carga todo; los centavos sobrantes no se pierden (Tarea 4).
4. **Notas de crédito y canceladas** en el mes y cuenta correctos (Tarea 7, `facturasAGastos`).
5. **Mes incompleto:** días sin ventas importadas se avisan y no rompen los % (Tarea 5 con venta 0 → `pct` null).

```sql
-- Pruebas de la migración 0025 (P&L fase 1). Córrelas completas en Supabase → SQL Editor después de la migración.
-- Necesitan dos usuarios activos de la misma sucursal y un usuario corporativo activo.
-- El resultado aparece como ERROR A PROPÓSITO para que Postgres deshaga todo:
--   · "PRUEBAS OK — …" → todo bien.   · "FALLÓ: …" → dice qué regla no se cumplió.

do $$
declare
  v_a uuid; v_b uuid; v_c uuid; v_suc uuid; v_prov uuid; v_fac uuid; v_n int;
  v_ok text[] := '{}';
begin
  select a.id, b.id, a.sucursal_id into v_a, v_b, v_suc
    from perfiles a join perfiles b on b.id <> a.id and b.activo and b.sucursal_id = a.sucursal_id
   where a.activo and a.sucursal_id is not null limit 1;
  if v_a is null then raise exception 'FALTAN DATOS: se necesitan dos usuarios activos de la misma sucursal.'; end if;
  select id into v_c from perfiles where rol::text = 'corporativo' and activo limit 1;
  if v_c is null then raise exception 'FALTAN DATOS: se necesita un usuario corporativo activo.'; end if;

  -- Precargas.
  if (select valor from pl_parametros where clave = 'rappi_pct' and vigente_desde = '2027-01-01') <> 0.18 then raise exception 'FALLÓ: Rappi 18 % desde 2027'; end if;
  if not exists (select 1 from nomina_plantilla where sucursal_id = v_suc and puesto = 'Gerente') then raise exception 'FALLÓ: plantilla precargada'; end if;
  if (select count(*) from pl_metas where sucursal_id = v_suc) <> 15 then raise exception 'FALLÓ: 15 metas por sucursal'; end if;
  v_ok := array_append(v_ok, 'parámetros, plantilla y metas precargados');

  -- La factura hereda la cuenta del proveedor.
  insert into proveedores (razon_social, rfc, regimen_fiscal, cuenta_pl) values ('Prueba PL', 'PPL010203AB1', '601', 'servicios') returning id into v_prov;
  insert into facturas (uuid, tipo, fecha_emision, rfc_emisor, nombre_emisor, regimen_emisor, rfc_receptor, subtotal, total, proveedor_id, vence_el, xml_ruta)
  values ('11111111-2222-3333-4444-555555555555', 'ingreso', current_date, 'PPL010203AB1', 'Prueba PL', '601', 'XAXX010101000', 100, 116, v_prov, current_date, 'prueba.xml')
  returning id into v_fac;
  if (select cuenta_pl from facturas where id = v_fac) <> 'servicios' then raise exception 'FALLÓ: la factura hereda la cuenta'; end if;
  v_ok := array_append(v_ok, 'la factura hereda la cuenta del proveedor');

  -- Gerente no ve; corporativo ve y no escribe; admin escribe.
  update perfiles set rol = 'gerente' where id = v_a;
  perform set_config('request.jwt.claims', json_build_object('sub', v_a, 'role', 'authenticated')::text, true);
  set local role authenticated;
  select count(*) into v_n from nomina_plantilla;
  reset role;
  if v_n <> 0 then raise exception 'FALLÓ: el gerente no debe ver la nómina'; end if;

  perform set_config('request.jwt.claims', json_build_object('sub', v_c, 'role', 'authenticated')::text, true);
  set local role authenticated;
  select count(*) into v_n from pl_metas;
  begin
    insert into gastos_mes (sucursal_id, mes, cuenta, monto, concepto) values (v_suc, date '2026-10-01', 'renta', 1, 'x');
    reset role;
    raise exception 'FALLÓ: corporativo no debe capturar';
  exception when others then
    reset role;
    if sqlerrm like 'FALLÓ%' then raise; end if;
    if sqlerrm not like '%row-level security%' then raise exception 'FALLÓ: mensaje inesperado (corporativo): %', sqlerrm; end if;
  end;
  if v_n = 0 then raise exception 'FALLÓ: corporativo debe ver las metas'; end if;

  update perfiles set rol = 'admin' where id = v_a;
  perform set_config('request.jwt.claims', json_build_object('sub', v_a, 'role', 'authenticated')::text, true);
  set local role authenticated;
  insert into gastos_mes (sucursal_id, mes, cuenta, monto, concepto) values (v_suc, date '2026-10-01', 'renta', 30000, 'Renta sin CFDI');
  insert into nomina_ajustes (sucursal_id, mes, renglon, monto, concepto) values (v_suc, date '2026-10-01', 'cocina', 1500, 'Bono');
  reset role;
  v_ok := array_append(v_ok, 'admin captura, corporativo solo ve, gerente no ve');

  -- Reglas de captura.
  begin
    insert into gastos_mes (sucursal_id, mes, cuenta, monto, concepto) values (v_suc, date '2026-10-15', 'renta', 1, 'x');
    raise exception 'FALLÓ: el mes es el día 1';
  exception when others then
    if sqlerrm like 'FALLÓ%' then raise; end if;
    if sqlerrm not like '%check constraint%' then raise exception 'FALLÓ: mensaje inesperado (mes): %', sqlerrm; end if;
  end;
  begin
    insert into gastos_mes (sucursal_id, mes, cuenta, monto, concepto) values (v_suc, date '2026-10-01', 'insumos', 1, 'x');
    raise exception 'FALLÓ: insumos no es cuenta de gasto del mes';
  exception when others then
    if sqlerrm like 'FALLÓ%' then raise; end if;
    if sqlerrm not like '%check constraint%' then raise exception 'FALLÓ: mensaje inesperado (cuenta): %', sqlerrm; end if;
  end;
  begin
    insert into pl_parametros (clave, valor, vigente_desde) values ('rappi_pct', 1.5, '2027-02-01');
    raise exception 'FALLÓ: un porcentaje va de 0 a 1';
  exception when others then
    if sqlerrm like 'FALLÓ%' then raise; end if;
    if sqlerrm not like '%check constraint%' then raise exception 'FALLÓ: mensaje inesperado (pct): %', sqlerrm; end if;
  end;
  v_ok := array_append(v_ok, 'mes el día 1, cuentas y porcentajes válidos');

  -- Cambiar la cuenta de una factura: solo admin.
  perform asignar_cuenta_factura(v_fac, 'mantenimiento');
  if (select cuenta_pl from facturas where id = v_fac) <> 'mantenimiento' then raise exception 'FALLÓ: el admin cambia la cuenta'; end if;
  perform set_config('request.jwt.claims', json_build_object('sub', v_c, 'role', 'authenticated')::text, true);
  begin
    perform asignar_cuenta_factura(v_fac, 'renta');
    raise exception 'FALLÓ: corporativo no cambia la cuenta';
  exception when others then
    if sqlerrm like 'FALLÓ%' then raise; end if;
    if sqlerrm not like 'Corporativo solo puede consultar%' then raise exception 'FALLÓ: mensaje inesperado: %', sqlerrm; end if;
  end;
  v_ok := array_append(v_ok, 'solo el admin cambia la cuenta de una factura');

  raise exception 'PRUEBAS OK — %', array_to_string(v_ok, ' · ');
end $$;
```

```ts
/** Cuentas del P&L: renglones que se alimentan de gastos (facturas, caja chica, gastos del mes) y memo. */
export const CUENTAS_GASTO = ["repartidores", "marketing_agregadores", "marketing", "renta", "servicios", "mantenimiento", "consumibles_equipo", "otros"] as const;
/** No entran al EBITDA: insumos (el alimento sale del costo teórico en fase 1) e inversiones (activo fijo). */
export const CUENTAS_MEMO = ["insumos", "inversiones"] as const;
export type CuentaGasto = (typeof CUENTAS_GASTO)[number];
export type CuentaMemo = (typeof CUENTAS_MEMO)[number];
export type CuentaPl = CuentaGasto | CuentaMemo;

export const NOMBRE_CUENTA: Record<CuentaPl, string> = {
  repartidores: "Repartidores", marketing_agregadores: "Marketing de agregadores", marketing: "Marketing",
  renta: "Renta", servicios: "Servicios (luz, gas, agua, internet)", mantenimiento: "Mantenimiento",
```

```ts
/** Nómina del P&L: plantilla por sucursal (null = oficina central) con vigencia, más ajustes del mes. */
export type RenglonNomina = "cocina" | "gerencia";
export type Puesto = {
  sucursal_id: string | null; puesto: string; renglon: RenglonNomina;
  personas: number; sueldo: number; pct_cargas: number; bono: number; vigente_desde: string;
};
export type Ajuste = { sucursal_id: string | null; renglon: RenglonNomina; monto: number; concepto: string };

const centavos = (n: number) => Math.round(n * 100) / 100;

/** Personas × (sueldo × (1 + cargas) + bono). */
export const costoPuesto = (p: Puesto) => centavos(Number(p.personas) * (Number(p.sueldo) * (1 + Number(p.pct_cargas)) + Number(p.bono)));

/** Por cada puesto, su versión vigente al cierre del mes (vigente_desde ≤ fin de mes, la más reciente). */
export function plantillaVigente(puestos: readonly Puesto[], sucursal: string | null, finDeMes: string): Puesto[] {
  const porPuesto = new Map<string, Puesto>();
  for (const p of puestos) {
    if (p.sucursal_id !== sucursal || p.vigente_desde > finDeMes) continue;
    const k = p.puesto.trim().toLowerCase();
    const previo = porPuesto.get(k);
    if (!previo || p.vigente_desde > previo.vigente_desde) porPuesto.set(k, p);
  }
  return [...porPuesto.values()].filter((p) => Number(p.personas) > 0);
}

export function nominaMes(puestos: readonly Puesto[], ajustes: readonly Ajuste[], sucursal: string | null, finDeMes: string) {
  const plantilla = { cocina: 0, gerencia: 0 };
  for (const p of plantillaVigente(puestos, sucursal, finDeMes)) plantilla[p.renglon] = centavos(plantilla[p.renglon] + costoPuesto(p));
  const ajuste = { cocina: 0, gerencia: 0 };
  for (const a of ajustes) if (a.sucursal_id === sucursal) ajuste[a.renglon] = centavos(ajuste[a.renglon] + Number(a.monto));
  return { plantilla, cocina: centavos(plantilla.cocina + ajuste.cocina), gerencia: centavos(plantilla.gerencia + ajuste.gerencia) };
}
```

```ts
/** Comisiones (Rappi, cobro) y regalías del P&L, día por día con el parámetro vigente ese día. */
import type { ClaveParametro } from "./parametros";

export type DiaVenta = {
  fecha: string; total: number; iva: number; tarjeta: number; en_linea: number;
  /** Lo del canal agregador (Rappi) ese día, con IVA: su cobro no pasa por dLocal ni NetPay. */
  agregador: { total: number; tarjeta: number; en_linea: number };
};
export type Vigente = (clave: ClaveParametro, fecha: string) => number;

const centavos = (n: number) => Math.round(n * 100) / 100;

export const comisionRappi = (dias: readonly DiaVenta[], vigente: Vigente) =>
  centavos(dias.reduce((s, d) => s + Number(d.agregador.total) * vigente("rappi_pct", d.fecha), 0));

export function comisionesCobro(dias: readonly DiaVenta[], vigente: Vigente) {
  let dlocal = 0, netpay = 0;
  for (const d of dias) {
    const enLinea = Math.max(0, Number(d.en_linea) - Number(d.agregador.en_linea));
    const tarjeta = Math.max(0, Number(d.tarjeta) - Number(d.agregador.tarjeta));
    const oxxo = vigente("oxxo_participacion", d.fecha), amex = vigente("amex_participacion", d.fecha);
    dlocal += enLinea * ((1 - oxxo) * vigente("dlocal_pct", d.fecha) + oxxo * vigente("oxxo_pct", d.fecha));
    netpay += tarjeta * ((1 - amex) * vigente("netpay_pct", d.fecha) + amex * vigente("amex_pct", d.fecha));
  }
  return { dlocal: centavos(dlocal), netpay: centavos(netpay), total: centavos(centavos(dlocal) + centavos(netpay)) };
}

export const regalias = (dias: readonly DiaVenta[], vigente: Vigente) =>
  centavos(dias.reduce((s, d) => s + (Number(d.total) - Number(d.iva)) * vigente("regalias_pct", d.fecha), 0));
```

```ts
/**
 * Reparte un total (oficina central) entre sucursales en proporción a su venta neta.
 * Sin ventas no reparte (todo en 0: queda solo en el consolidado). Los centavos que sobran van a la de más venta.
 */
export function prorratear(total: number, ventas: ReadonlyMap<string, number>): Map<string, number> {
  const suma = [...ventas.values()].reduce((s, v) => s + Math.max(0, v), 0);
  const r = new Map<string, number>();
  if (suma <= 0) { for (const k of ventas.keys()) r.set(k, 0); return r; }
  let asignado = 0, mayor: string | null = null;
  for (const [k, v] of ventas) {
    const parte = Math.floor((total * Math.max(0, v) / suma) * 100) / 100;
    r.set(k, parte);
    asignado += parte;
    if (mayor === null || v > (ventas.get(mayor) ?? 0)) mayor = k;
  }
  const resto = Math.round((total - asignado) * 100) / 100;
  if (mayor !== null && resto) r.set(mayor, Math.round((r.get(mayor)! + resto) * 100) / 100);
  return r;
}
```

```ts
  if (meta.tipo === "pct_rappi") return centavos(ctx.ventaRappi * v);

export function calcularPL(e: EntradaPL, metas: readonly Meta[], ctx: ContextoMeta): FilaPL[] {

  const montos: Partial<Record<ClaveRenglon, number | null>> = {
    venta_neta: e.ventaNeta, alimento: e.alimento, merma: e.merma,
    nomina_cocina: e.nomina.cocina, nomina_gerencia: e.nomina.gerencia,

    consumibles_equipo: g("consumibles_equipo"), regalias: e.regalias,
    otros: centavos(g("otros") + e.sinCuenta + e.comisionesCobro + e.faltantes), oficina: e.oficina,
  };
  const gastosTienda = RENGLONES.filter((r) => r.tipo === "gasto" && r.clave !== "oficina");
  const sumaTienda = gastosTienda.reduce((s, r) => s + (montos[r.clave] ?? 0), 0);
  montos.ebitda_tienda = centavos(e.ventaNeta - sumaTienda);
  montos.ebitda = centavos(montos.ebitda_tienda - e.oficina);

  const metaDe = new Map(metas.map((m) => [m.renglon, metaMonto(m, ctx)]));
  const metasTienda = gastosTienda.map((r) => metaDe.get(r.clave));
  const hayMetas = metas.length > 0;
  const metaTienda = hayMetas ? centavos(ctx.ventaNeta - metasTienda.reduce<number>((s, m) => s + (m ?? 0), 0)) : null;
  metaDe.set("ebitda_tienda", metaTienda as number);
  if (metaTienda !== null) metaDe.set("ebitda", centavos(metaTienda - (metaDe.get("oficina") ?? 0)));
  if (hayMetas) metaDe.set("venta_neta", ctx.ventaNeta);

  const pct = (n: number | null | undefined) => (n === null || n === undefined || e.ventaNeta <= 0 ? null : Math.round((n / e.ventaNeta) * 10000) / 10000);
  return RENGLONES.map((r) => {
    const monto = montos[r.clave] ?? null;
    const meta = metaDe.has(r.clave) ? metaDe.get(r.clave) ?? null : null;
    return {
      clave: r.clave, nombre: r.nombre, tipo: r.tipo, monto, pct: pct(monto), meta,
      metaPct: meta === null || ctx.ventaNeta <= 0 ? null : Math.round((meta / ctx.ventaNeta) * 10000) / 10000,
      diferencia: meta === null || monto === null ? null : centavos(monto - meta),
    };
  });
}

/** Metas del consolidado: por renglón, la suma de la meta en pesos de cada sucursal (como monto fijo). */

}
```

```ts
export function facturasAGastos(facturas: readonly { tipo: "ingreso" | "egreso"; estado: string; subtotal: number; descuento: number; cuenta_pl: string | null }[]) {
  const gastos: Partial<Record<CuentaGasto, number>> = {};
  let sinCuenta = 0;
  const memos = { insumos: 0, inversiones: 0 };
  for (const f of facturas) {
    if (f.estado === "cancelada") continue;
    const monto = (Number(f.subtotal) - Number(f.descuento)) * (f.tipo === "egreso" ? -1 : 1);
    if (f.cuenta_pl === "insumos" || f.cuenta_pl === "inversiones") memos[f.cuenta_pl] = centavos(memos[f.cuenta_pl] + monto);
    else if (f.cuenta_pl && GASTO.includes(f.cuenta_pl)) gastos[f.cuenta_pl as CuentaGasto] = centavos((gastos[f.cuenta_pl as CuentaGasto] ?? 0) + monto);
    else sinCuenta = centavos(sinCuenta + monto);
  }
  return { gastos, sinCuenta, memos };
}

export function sumarGastos(...partes: Partial<Record<CuentaGasto, number>>[]): Partial<Record<CuentaGasto, number>> {
  const r: Partial<Record<CuentaGasto, number>> = {};
  for (const p of partes) for (const [k, v] of Object.entries(p) as [CuentaGasto, number][]) r[k] = centavos((r[k] ?? 0) + v);
  return r;
}

/** Faltantes de efectivo como gasto: −Σ diferencias (una diferencia negativa es faltante). */
export const faltantesDe = (diferencias: readonly number[]) => centavos(-diferencias.reduce((s, d) => s + Number(d), 0));
```

```ts
    };

  });

  const manualOf = de(gastosMes, null).reduce((t, g) => t + Number(g.monto), 0);
  const oficina = centavos(Object.values(fOf.gastos).reduce((t, v) => t + v, 0) + fOf.sinCuenta + nOf.cocina + nOf.gerencia + manualOf);

  }

  entrada.memos = { insumos: centavos(entrada.memos.insumos + fOf.memos.insumos), inversiones: centavos(entrada.memos.inversiones + fOf.memos.inversiones) };
  const ctx: ContextoMeta = {
    ventaNeta: entrada.ventaNeta,
    ventaRappi: centavos(porSucursal.reduce((t, s) => t + s.ctx.ventaRappi, 0)),
    plantilla: {
      cocina: centavos(porSucursal.reduce((t, s) => t + s.ctx.plantilla.cocina, 0)),
      gerencia: centavos(porSucursal.reduce((t, s) => t + s.ctx.plantilla.gerencia, 0)),
    },
  };
  return { mes, porSucursal, oficina, consolidado: { entrada, ctx }, detalle };
}

/** P&L de una sucursal (o consolidado si no hay o no existe) en un mes ya cargado. */
export function vistaPL(d: DatosMes, sucursalId: string | undefined) {
  const suc = d.porSucursal.find((s) => s.id === sucursalId) ?? null;
  if (suc) return { suc, titulo: suc.nombre, entrada: suc.entrada, filas: calcularPL(suc.entrada, suc.metas, suc.ctx), avisos: suc.avisos };
  return {
    suc: null, titulo: "Consolidado", entrada: d.consolidado.entrada,
    filas: calcularPL(d.consolidado.entrada, sumarMetas(d.porSucursal), d.consolidado.ctx),
    avisos: d.porSucursal.flatMap((s) => s.avisos.map((a) => `${s.nombre}: ${a}`)),
  };
}

/** El mes elegido con su mes anterior (% de venta) y el acumulado del año (% de venta), para pantalla, PDF y Excel. */
export async function plConHistoria(supabase: SupabaseClient, mes: string, sucursalId: string | undefined) {
  const acumulados = mesesAcumulados(mes);
  const anterior = mesAnterior(mes);
  const meses = [...new Set([...acumulados, ...(anterior >= INICIO_PL ? [anterior] : [])])];
  const cargados = new Map(await Promise.all(meses.map(async (m) => [m, await datosPL(supabase, m)] as const)));
  const d = cargados.get(mes)!;
  const vista = vistaPL(d, sucursalId);
  const previo = cargados.get(anterior);
  const entradaDe = (x: DatosMes) => vistaPL(x, sucursalId).entrada;
  const acumulado = sumarEntradas(acumulados.map((m) => entradaDe(cargados.get(m)!)));
  const sinMetas: ContextoMeta = { ventaNeta: acumulado.ventaNeta, ventaRappi: 0, plantilla: { cocina: 0, gerencia: 0 } };
  const pctDe = (filas: FilaPL[]) => new Map(filas.map((f) => [f.clave, f.pct]));
  return {
    datos: d, ...vista,
    anterior: previo ? pctDe(vistaPL(previo, sucursalId).filas) : null,
    acumulado: pctDe(calcularPL(acumulado, [], sinMetas)),
    detalleDe: (renglon: string) => (vista.suc ? d.detalle.get(`${vista.suc.id}|${renglon}`) ?? [] : []),
    memoDetalle: (cuenta: "insumos" | "inversiones") => (vista.suc ? d.detalle.get(`${vista.suc.id}|memo_${cuenta}`) ?? [] : []),
  };
}

/** Mes de la URL si es válido (de INICIO_PL al mes actual); si no, el mes actual. */
export const mesElegido = (m: string | undefined) => {
  const hoy = hoyMX().slice(0, 7);
  return m && /^\d{4}-\d{2}$/.test(m) && m <= hoy && m >= INICIO_PL ? m : hoy;
};

export const NOMBRE_MEMO: Record<"insumos" | "inversiones", string> = { insumos: NOMBRE_CUENTA.insumos, inversiones: NOMBRE_CUENTA.inversiones };
```

```ts
const RUTA = "/finanzas/nomina";
const t = (fd: FormData, c: string) => String(fd.get(c) ?? "").trim();
const sucursal = (v: string) => (v === "oficina" || !v ? null : v);

/** Alta o cambio de un puesto: se guarda como versión nueva con su fecha de vigencia (personas 0 = se quita). */
export async function guardarPuesto(fd: FormData) {
  const supabase = await clienteFinanzas(RUTA);
  const puesto = t(fd, "puesto"), renglon = t(fd, "renglon"), vigente_desde = t(fd, "vigente_desde");
  const personas = Math.round(numero(fd.get("personas"))), sueldo = numero(fd.get("sueldo")), bono = numero(fd.get("bono"));
  const pct_cargas = numero(fd.get("pct_cargas")) / 100;
  if (!puesto) volver(RUTA, "error", "Escribe el puesto.");
  if (renglon !== "cocina" && renglon !== "gerencia") volver(RUTA, "error", "Elige si es cocina o gerencia.");
  if (!/^\d{4}-\d{2}-\d{2}$/.test(vigente_desde)) volver(RUTA, "error", "Elige desde cuándo aplica.");
  if (personas < 0 || sueldo < 0 || bono < 0 || pct_cargas < 0 || pct_cargas > 1) volver(RUTA, "error", "Revisa personas, sueldo, cargas (0 a 100 %) y bono.");
  const { error } = await supabase.from("nomina_plantilla").insert(
    { sucursal_id: sucursal(t(fd, "sucursal_id")), puesto, renglon, personas, sueldo, pct_cargas, bono, vigente_desde },
  );
  if (error) volver(RUTA, "error", error.code === "23505" ? "Ya hay una versión de ese puesto con esa fecha; usa otra fecha." : error.message);
  revalidatePath("/finanzas", "layout");
  volver(RUTA, "ok", "Puesto guardado.");
}

export async function agregarAjuste(fd: FormData) {
  const supabase = await clienteFinanzas(RUTA);
  const mes = t(fd, "mes"), renglon = t(fd, "renglon"), concepto = t(fd, "concepto");
  const monto = numero(fd.get("monto"));

  if (renglon !== "cocina" && renglon !== "gerencia") volver(RUTA, "error", "Elige si es cocina o gerencia.");
  if (!concepto || !monto) volver(RUTA, "error", "Escribe el concepto y el monto (negativo si resta).");
  const { error } = await supabase.from("nomina_ajustes").insert({ sucursal_id: sucursal(t(fd, "sucursal_id")), mes: `${mes}-01`, renglon, monto, concepto });
  if (error) volver(RUTA, "error", error.message);
  revalidatePath("/finanzas", "layout");
  volver(RUTA, "ok", "Ajuste agregado.");

export async function quitarAjuste(fd: FormData) {
```

```ts
  const clave = t(fd, "clave"), vigente_desde = t(fd, "vigente_desde");
  const valor = numero(fd.get("valor")) / 100;
  if (!CLAVES.includes(clave)) volver(RUTA, "error", "Elige el parámetro.");
  if (!/^\d{4}-\d{2}-\d{2}$/.test(vigente_desde)) volver(RUTA, "error", "Elige desde cuándo aplica.");
  if (valor < 0 || valor > 1) volver(RUTA, "error", "El porcentaje va de 0 a 100.");
  const { error } = await supabase.from("pl_parametros").upsert({ clave, valor, vigente_desde }, { onConflict: "clave,vigente_desde" });
  if (error) volver(RUTA, "error", error.message);
  revalidatePath("/finanzas", "layout");
  volver(RUTA, "ok", "Parámetro guardado.");
}

/** Meta de un renglón de una sucursal: % de venta, monto fijo, % de venta Rappi o plantilla. */
export async function guardarMeta(fd: FormData) {
  const supabase = await clienteFinanzas(RUTA);
  const sucursal_id = t(fd, "sucursal_id"), renglon = t(fd, "renglon"), tipo = t(fd, "tipo");
  const crudo = numero(fd.get("valor"));
  if (!RENGLONES.includes(renglon) || !["pct", "monto", "pct_rappi", "plantilla"].includes(tipo)) volver(RUTA, "error", "Revisa el renglón y el tipo de meta.");
  const valor = tipo === "pct" || tipo === "pct_rappi" ? crudo / 100 : crudo;
  if (valor < 0) volver(RUTA, "error", "La meta no puede ser negativa.");
  const { error } = await supabase.from("pl_metas").upsert({ sucursal_id, renglon, tipo, valor }, { onConflict: "sucursal_id,renglon" });
  if (error) volver(RUTA, "error", error.message);
  revalidatePath("/finanzas", "layout");
  volver(`${RUTA}?sucursal=${sucursal_id}`, "ok", "Meta guardada.");
}
```

```ts
import type { Celda, HojaXlsx } from "../xlsx-escribir";
import type { FilaPL } from "./pl";

const pesos = (n: number | null): Celda => (n === null ? null : { valor: n, formato: "pesos" });
const pct = (n: number | null | undefined): Celda => (n === null || n === undefined ? null : { valor: n, formato: "porcentaje" });

/** El P&L del mes en una hoja de Excel (montos como números con formato, % reales). */
export function hojaPL(r: {
  titulo: string; mes: string; generado: string; filas: FilaPL[];
  anterior: ReadonlyMap<string, number | null> | null; acumulado: ReadonlyMap<string, number | null>;
}): HojaXlsx[] {
  const filas: Celda[][] = [
    [`P&L · ${r.titulo}`], [`Mes ${r.mes} · sin IVA, criterio devengado`], [`Generado por ${r.generado}`], [],
    ["Renglón", "Real", "% de venta", "Meta", "Meta %", "Diferencia", "Mes anterior %", "Acumulado %"],
    ...r.filas.map((f): Celda[] => [
      f.nombre, f.monto === null ? "Sin dato" : pesos(f.monto), pct(f.pct), pesos(f.meta), pct(f.metaPct), pesos(f.diferencia),
      r.anterior ? pct(r.anterior.get(f.clave)) : null, pct(r.acumulado.get(f.clave)),
    ]),
  ];
  const negritas = [0, 4, ...r.filas.flatMap((f, i) => (f.tipo === "total" || f.tipo === "venta" ? [i + 5] : []))];
  return [{ nombre: "P&L", filas, negritas, anchos: [30, 16, 12, 16, 12, 16, 16, 14] }];
}
```
